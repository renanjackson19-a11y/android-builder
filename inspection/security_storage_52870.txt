// Metadados lidos pelo painel GreenPlay ao clicar em "Detectar versão".
// versionName/versionCode vêm sempre do Gradle; as correções vêm do arquivo
// app/RELEASE_NOTES.txt e são incorporadas no APK automaticamente.
def gpMetadataTask = tasks.register('generateGreenPlayUpdateMetadata') {
    doLast {
        def assetsDir = file('src/main/assets')
        if (!assetsDir.exists()) assetsDir.mkdirs()
        def releaseFile = file('RELEASE_NOTES.txt')
        def notes = releaseFile.exists()
                ? releaseFile.readLines('UTF-8').collect { it.trim() }.findAll { !it.isEmpty() && !it.startsWith('#') }
                : []
        def payload = [
                package    : android.defaultConfig.applicationId,
                versionName: android.defaultConfig.versionName,
                versionCode: android.defaultConfig.versionCode,
                notes      : notes
        ]
        new File(assetsDir, 'greenplay_update.json').setText(JsonOutput.prettyPrint(JsonOutput.toJson(payload)), 'UTF-8')
        new File(assetsDir, 'greenplay_build.properties').setText(
                'package=' + android.defaultConfig.applicationId + '\n' +
                'versionName=' + android.defaultConfig.versionName + '\n' +
                'versionCode=' + android.defaultConfig.versionCode + '\n', 'UTF-8')
    }
}

afterEvaluate {
    tasks.named('preBuild').configure { dependsOn(gpMetadataTask) }
}
